Swiper 官方 effect 只有 6 种（slide / fade / cube / coverflow / flip / cards），还藏着第 7 种——creative。 它让你用 transform 自己写 prev / next slide 怎么出场。下面 7 个框：前 6 个是课堂上讲的 6 种效果，第 7 个是用 creative 自学的电影墙， 按住图片左右拖动就能看到对应的切换动画。

任务二 · JS 核心代码 
// 配方表：前 6 项为课堂 6 种 effect，第 7 项为 creative 电影墙（AI 自学）
var zyx1029_recipes = [
  { id: '#zyx1029C1', effect: 'slide' },     // ① 平移：默认效果，slide 跟手水平滑动
  { id: '#zyx1029C2', effect: 'fade' },      // ② 淡入淡出：原地透明度渐变，无位移
  { id: '#zyx1029C3', effect: 'cube' },      // ③ 立方体：绕 Y 轴 3D 旋转翻面
  { id: '#zyx1029C4', effect: 'coverflow' }, // ④ 封面流：中间清晰、两侧缩小带透视
  { id: '#zyx1029C5', effect: 'flip' },      // ⑤ 翻牌：沿边缘 3D 翻转 180 度
  { id: '#zyx1029C6', effect: 'cards' },     // ⑥ 卡片堆叠：像扑克牌叠放、顶层滑出
  { id: '#zyx1029C7', effect: 'creative',    // ⑦ 自定义电影墙（第 7 种，课堂未讲，AI 自学）
    creativeEffect: {
      prev: { translate: ['-80%', '-10%', -300], rotate: [0, 0, -20], scale: 0.7 },
      next: { translate: ['110%', '10%', 0],     rotate: [0, 0, 10],  scale: 1.2 }
    } }
];

// 遍历配方表逐个初始化：loop 循环 + 600ms 动画 + 配方表的 effect / creativeEffect
// 前 6 项的 creativeEffect 为 undefined，Swiper 自动忽略——一段代码通吃 7 种效果
zyx1029_recipes.forEach(function (zyx1029R) {
  new Swiper(zyx1029R.id, {
    loop: true,
    speed: 600,
    effect: zyx1029R.effect,
    creativeEffect: zyx1029R.creativeEffect
  });
});
