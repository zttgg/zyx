任务三 · JS 核心代码 
var zyx1029SwT3 = new Swiper('#zyx1029SwT3', {
  // loop 无缝循环：最后一张的"下一张"自动接回第一张，轮播可一直向同方向翻
  loop: true,
  // speed 切换动画时长（毫秒）：每次翻页用 800ms，动画比默认 300ms 更从容
  speed: 800,
  // keyboard 键盘控制（超纲新增；🟥 老师红笔圈画 keyboard/mousewheel——必问为什么选 keyboard）：
  //   true = 只在轮播获得焦点/鼠标悬停时，按 ← 相当于 slidePrev()、按 → 相当于 slideNext()
  keyboard: true,
  // autoplay 自动播放：3.2 秒一张；disableOnInteraction:false = 手动操作后不停（任务一关键考点）
  autoplay: { delay: 3200, disableOnInteraction: false },
  // pagination 圆点分页器：el 挂载空 div 自动生成 6 个圆点；clickable 可点击直达
  pagination: { el: '#zyx1029PagT3', clickable: true }
});
